from sklearn.datasets import fetch_20newsgroups

from sklearn.feature_extraction.text import CountVectorizer

from sklearn.naive_bayes import MultinomialNB, BernoulliNB

from sklearn.metrics import accuracy_score, f1_score



cats = ['sci.space', 'rec.autos']



train = fetch_20newsgroups(

    subset='train',

    categories=cats

)



test = fetch_20newsgroups(

    subset='test',

    categories=cats

)



vec_multi = CountVectorizer()



X_train_m = vec_multi.fit_transform(train.data)

X_test_m = vec_multi.transform(test.data)



vec_bern = CountVectorizer(binary=True)
X_train_b = vec_bern.fit_transform(train.data)

X_test_b = vec_bern.transform(test.data)



m_nb = MultinomialNB().fit(

    X_train_m,

    train.target

)



b_nb = BernoulliNB().fit(

    X_train_b,

    train.target

)



print(f"Multinomial Accuracy: {accuracy_score(test.target, m_nb.predict(X_test_m)):.2f}")

print(f"Bernoulli Accuracy: {accuracy_score(test.target, b_nb.predict(X_test_b)):.2f}")

/**************************************************************************************/

OUTPUT
Multinomial Accuracy: 0.99
Bernoulli Accuracy: 0.93
